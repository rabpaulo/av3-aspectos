int factorial(int n) {
    int result = 1;
    while (n > 1) {
        result = result * n;
        n = n - 1;
    }
    return result;
}

int main() {
    int x = 5;
    float y = 3.14;
    if (x == 5) {
        x = factorial(x);
    } else {
        x = 0;
    }
    return x;
}
